true | false | true; echo $?
#---
# control: background and $!
true & p=$!; wait $p; echo $?; [ -n "$p" ] && echo set
#---
# control: wait with no children
wait; echo $?
#---
# control: wait for specific failing job
(exit 4) & wait $!; echo $?
#---
# control: wait unknown pid
wait 99999 2>/dev/null; echo $?
#---
# control: here-doc as stdin of background
cat <<EOF & wait
bg
EOF
#---
# function: define and call
f() { echo "in $1"; }; f x
#---
# function: return status
f() { return 3; }; f; echo $?
#---
# function: return without value
f() { false; return; }; f; echo $?
#---
# function: recursion
f() { [ $1 -le 0 ] && return 0; echo $1; f $(($1-1)); }; f 3
#---
# function: positional isolation
f() { echo $#; }; set -- a b c; f x; echo $#
#---
# function: shift in a function
f() { shift; echo $@; }; set -- a b; f 1 2 3; echo $@
#---
# function: variables are global
f() { x=2; }; x=1; f; echo $x
#---
# function: function with redirection
f() { echo hi; } > f.out; f; cat f.out
#---
# function: function body is a compound command
f() ( echo sub; x=1 ); f; echo "[$x]"
#---
# function: function with if body
f() if true; then echo y; fi; f
#---
# function: redefinition
f() { echo 1; }; f() { echo 2; }; f
#---
# function: unset
f() { echo 1; }; unset -f f; f 2>/dev/null; echo $?
#---
# function: call before definition
f 2>/dev/null; echo $?; f() { echo x; }; f
#---
# function: function overrides a regular builtin
echo() { printf 'fn %s\n' "$*"; }; echo hi; command echo hi
#---
# function: function overriding a special builtin
eval() { echo fn; }; eval x
#---
# function: function named like a special builtin and exit
export() { echo fn; }; export x 2>/dev/null; echo $?
#---
# function: function and assignment prefix
f() { echo "[$a]"; }; a=1 f; echo "[$a]"
#---
# function: function and exit
f() { exit 3; }; f; echo no
#---
# function: function in subshell and return
f() { return 2; }; (f; echo $?)
#---
# function: return in dot script inside function
echo 'return 5' > s; f() { . ./s; echo after; }; f; echo $?
#---
# function: return in dot at top level
echo 'return 5; echo no' > s; . ./s; echo $?
#---
# function: return in a loop in a function
f() { for i in 1 2; do return 4; done; }; f; echo $?
#---
# function: break in a function inside a loop
f() { break; }; for i in 1 2; do f; echo $i; done; echo end
#---
# function: local variables via subshell
f() ( x=2; echo $x ); x=1; f; echo $x
#---
# function: function definition status
f() { :; }; echo $?
#---
# function: funcs with the same name as variable
f=1; f() { echo fn; }; f; echo $f
#---
# function: $0 inside
f() { echo "${0##*/}" | head -c 0; echo ok; }; f
#---
# function: nested definition
f() { g() { echo g; }; }; f; g
#---
# function: recursion depth moderate
f() { [ $1 -gt 200 ] && echo done && return; f $(($1+1)); }; f 0
#---
# function: arguments with spaces
f() { for a in "$@"; do echo "[$a]"; done; }; f 'a b' c ''
#---
# function: "$@" empty argument kept
f() { echo $#; }; f "" ""
#---
# function: function call in command substitution
f() { echo sub; }; x=$(f); echo $x
#---
# function: function and trap
f() { trap 'echo t' EXIT; }; f; echo done
#---
# function: function with here-doc
f() { cat; }; f <<EOF
body
EOF
#---
# function: definition with newline before brace
f()
{
  echo nl
}
f
#---
# function: definition with space before parens
f ( ) { echo sp; }; f
#---
# function: definition with redirect on body
f() { cat; } <<EOF
fromdef
EOF
f
#---
# function: invalid function name
1f() { :; }; echo $?
#---
# function: function name with a dash
a-b() { echo dash; }; a-b
#---
# function: quoted function name
'f'() { echo q; }; f
